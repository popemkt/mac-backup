1. System Concept & Design Philosophy Core Design

kb is a repo-native outliner datastore built with Bun and TypeScript. Inspired by Tana and Logseq, it models all user data, fields, supertags, and queries as first-class Nodes in a unified graph.

Key Decision: Everything is a Node
Fields and Supertags are not secondary relational schemas—they are nodes (typed sys.field and sys.tag). Attaching a field to a node means attaching a property keyed by the field node's ULID.

6. Harman Control Boundary & Action Registry

// Excerpt: Receipt safety boundary in tools/kb/src/registry.ts
export async function invoke(
  ctx: KbContext,
  invocation: ActionInvocation,
): Promise<ActionReceipt> {
  const { id, input } = invocation;
  try {
    const registry = await registryFor(ctx.root);
    const entry = registry.byId.get(id);
    if (!entry) return failed(id, "unknown_action", `unknown action: ${id}`);
    const parsed = entry.def.inputSchema.parse(input);
    const output = await entry.handler(ctx, parsed as never);
    return succeeded(id, output);
  } catch (err) {
    return receiptFromError(id, err); // Traps ZodError, ResolveError, Internal
  }
}

14. Optimistic Mutation Layer

Mutations apply instantly to the local store via pure transaction planners in actions/plan.ts. The UI then asynchronously dispatches the operation to POST /api/action. If the remote request fails, a rollback restores the prior snapshot.

// Excerpt: Optimistic flow in tools/kb/ui/src/actions/optimistic.ts
export async function runOptimistic(
  plan: PlannedMutation,
): Promise<{ ok: boolean; receipt?: ActionReceipt }> {
  const store = useOutlineStore.getState();
  const snapshot = cloneWireNodes(store.wireNodes);

  // Apply optimistic changes locally immediately
  store.applyTx(plan.upserts, plan.deletes);

  try {
    const receipt = await postAction(plan.actionId, plan.input);
    if (receipt.status === "failed") {
      // Revert store state on failure
      store.restoreSnapshot(snapshot, store.rev);
      return { ok: false, receipt };
    }
    return { ok: true, receipt };
  } catch (err) {
    store.restoreSnapshot(snapshot, store.rev);
    return { ok: false };
  }
}
